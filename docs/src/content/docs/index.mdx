Frameworks bundle reactivity, rendering, and lifecycle into one package deal. Swap one, rewrite them all.

ElementsKit unbundles them: signals, JSX, and custom elements as independent primitives — use one, two, or all three, inside React or on their own. No virtual DOM. No framework lock-in.

**Import what you use.** Every module is a separate subpath — `elements-kit/signals`, `elements-kit/utilities/media-query`, `elements-kit/integrations/react`. Bundlers tree-shake the rest.

```tsx twoslash
// @noErrors
import { signal, computed } from "elements-kit/signals";
import { render } from "elements-kit/render";
import type { ReactiveProps } from "elements-kit/jsx-runtime";

function Counter(props: ReactiveProps<{ initial?: number }>) {
  const count = signal(props.initial() ?? 0);
  const doubled = computed(() => count() * 2);

  return (
    <section>
      <p><strong>{count}</strong> × 2 = <strong>{doubled}</strong></p>
      <button on:click={() => count(count() + 1)}>+1</button>{" "}
      <button on:click={() => count(count() - 1)}>−1</button>
    </section>
  );
}

render(document.getElementById("app")!, () => <Counter initial={0} />);

// @noErrors
import { signal } from "elements-kit/signals";
import { async } from "elements-kit/utilities/async";
import { render } from "elements-kit/render";

const query = signal("hello");

const search = async(async () => {
  const res = await fetch(`/api/search?q=${query()}`);
  return res.json();
}).start();

render(document.getElementById("app")!, () => (
  <section>
    <input
      value={query}
      on:input={(e) => query((e.target as HTMLInputElement).value)}
    />
    <p>state: <strong>{() => search.state}</strong></p>
    <pre>{() => JSON.stringify(search.value, null, 2)}</pre>
  </section>
));
```

  <TabItem label="Custom element">
```tsx twoslash
// @noErrors
import { reactive, computed } from "elements-kit/signals";
import { attributes, ATTRIBUTES as attr } from "elements-kit/attributes";
import { render } from "elements-kit/render";

@attributes
class CounterElement extends HTMLElement {
  static [attr] = {
    initial(this: CounterElement, v: string | null) {
      this.count = Number(v ?? 0);
    },
  };

  @reactive() count = 0;
  doubled = computed(() => this.count * 2);

  #unmount?: () => void;

  #template = () => (
    <p>
      <strong>{() => this.count}</strong> × 2 = <strong>{this.doubled}</strong>{" "}
      <button on:click={() => this.count++}>+1</button>
    </p>
  );

  connectedCallback() {
    this.#unmount = render(this, this.#template);
  }

  disconnectedCallback() {
    this.#unmount?.();
    this.#unmount = undefined;
  }
}

customElements.define("x-counter", CounterElement);

// Usable anywhere — plain HTML, React, Vue:
// <x-counter initial="5"></x-counter>
```

  <TabItem label="Use in React.js">
```tsx twoslash
// @noErrors
/** @jsxImportSource react */

import { signal, computed, effect } from "elements-kit/signals";
import { useSignal, useScope } from "elements-kit/integrations/react";

// Signals live outside React — shared with any other consumer.
const count = signal(0);
const doubled = computed(() => count() * 2);

export default function Counter() {
  const value = useSignal(count);
  const double = useSignal(doubled);

  useScope(() => {
    effect(() => console.log("count:", count()));
  });

  return (
    <section>
      <p><strong>{value}</strong> × 2 = <strong>{double}</strong></p>
      <button onClick={() => count(count() + 1)}>+1</button>{" "}
      <button onClick={() => count(count() - 1)}>−1</button>
    </section>
  );
}
```
