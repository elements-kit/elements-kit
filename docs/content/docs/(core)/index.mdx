**ElementsKit is a library of reactive primitives.** Pick one, use it in plain JavaScript, React, a custom element, or on the server. Everything builds on one small core; JSX, custom elements and the UI Kit are modules you add only if you want them.

<Modules />

Every piece is its own import, so you ship only what you use.

- **Compose, don't configure.** Small focused APIs: `signal`, `computed`, `on`, `fromEvent`, `async`. Combine primitives instead of maintaining an overloaded interface.
- **Close to the platform.** `promise` extends `Promise`. Custom elements *are* `HTMLElement`. JSX compiles to `document.createElement`. No virtual DOM, no proxies.
- **Predictable and explicit, no magic.** Signals are reactive; nothing else is. No hidden subscriptions.
- **Designed for the AI age.** Code is cheap; maintenance still isn't. Swap one block at a time.

```ts twoslash
// @noErrors
import { signal, computed, effect } from "elements-kit/signals";
import { online } from "elements-kit/utilities/network";
import { createMediaQuery } from "elements-kit/utilities/media-query";
import { on } from "elements-kit/utilities/event-listener";

const clicks = signal(0);
const dark = createMediaQuery("(prefers-color-scheme: dark)");
const label = computed(() => `${clicks()} clicks · ${online() ? "online" : "offline"}`);

const button = document.querySelector("button")!;
on(button, "click", () => clicks(clicks() + 1));

effect(() => {
  button.textContent = label();
  document.body.classList.toggle("dark", dark());
});
```

```tsx twoslash
// @noErrors
/** @jsxImportSource react */

import { signal, computed, effect } from "elements-kit/signals";
import { useSignal, useScope } from "elements-kit/integrations/react";

// Signals live outside React, shared with any other consumer.
const count = signal(0);
const doubled = computed(() => count() * 2);

export default function Counter() {
  const value = useSignal(count);
  const double = useSignal(doubled);

  useScope(() => {
    effect(() => console.log("count:", count()));
  });

  return (
    <section>
      <p><strong>{value}</strong> × 2 = <strong>{double}</strong></p>
      <button onClick={() => count(count() + 1)}>+1</button>{" "}
      <button onClick={() => count(count() - 1)}>−1</button>
    </section>
  );
}
```

  <Tab value="Custom element">
```tsx twoslash
// @noErrors
import { reactive, computed } from "elements-kit/signals";
import { attributes, ATTRIBUTES as attr } from "elements-kit/attributes";
import { render } from "elements-kit/render";

@attributes
class CounterElement extends HTMLElement {
  static [attr] = {
    initial(this: CounterElement, v: string | null) {
      this.count = Number(v ?? 0);
    },
  };

  @reactive() count = 0;
  doubled = computed(() => this.count * 2);

  #unmount?: () => void;

  #template = () => (
    <p>
      <strong>{() => this.count}</strong> × 2 = <strong>{this.doubled}</strong>{" "}
      <button on:click={() => this.count++}>+1</button>
    </p>
  );

  connectedCallback() {
    this.#unmount = render(this, this.#template);
  }

  disconnectedCallback() {
    this.#unmount?.();
    this.#unmount = undefined;
  }
}

customElements.define("x-counter", CounterElement);

// Usable anywhere: plain HTML, React, Vue.
// <x-counter initial="5"></x-counter>
```

```tsx twoslash
// @noErrors
import { signal, computed, resolve } from "elements-kit/signals";
import { render } from "elements-kit/render";
import type { Props } from "elements-kit/jsx-runtime";

function Counter(props: Props<{ initial?: number }>) {
  const count = signal(resolve(props.initial) ?? 0);
  const doubled = computed(() => count() * 2);

  return (
    <section>
      <p><strong>{count}</strong> × 2 = <strong>{doubled}</strong></p>
      <button on:click={() => count(count() + 1)}>+1</button>{" "}
      <button on:click={() => count(count() - 1)}>−1</button>
    </section>
  );
}

render(document.getElementById("app")!, () => <Counter initial={0} />);
```

  <Tab value="Async">
```tsx twoslash
// @noErrors
import { signal } from "elements-kit/signals";
import { async } from "elements-kit/utilities/async";
import { render } from "elements-kit/render";

const query = signal("hello");

const search = async(async () => {
  const res = await fetch(`/api/search?q=${query()}`);
  return res.json();
}).start();

render(document.getElementById("app")!, () => (
  <section>
    <input
      value={query}
      on:input={(e) => query((e.target as HTMLInputElement).value)}
    />
    <p>state: <strong>{() => search.state}</strong></p>
    <pre>{() => JSON.stringify(search.value, null, 2)}</pre>
  </section>
));
```
